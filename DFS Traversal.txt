graph = {
    'A': ['B', 'C'],
    'B': [],
    'C': []
}

visited = set()

def dfs(node):
    if node not in visited:
        print(node, end=" ")
        visited.add(node)
        for neighbour in graph[node]:
            dfs(neighbour)
            
            
print("DFS Traversal:")
dfs('A')
